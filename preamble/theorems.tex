%======================================================================
%  theorems.tex — theorem-like environments.
%  All share the counter 'result', numbered within CHAPTER: results run
%  1.1, 1.2, ..., 1.n through Chapter 1, then restart at 2.1, and so on.
%
%  Numbering within *section* was the original scheme, but it collided
%  with the sectional numbering: "Proposition 1.1.36" and "Subsection
%  1.1.4" occupied the same number space and were told apart only by the
%  word in front of them. Per-chapter numbering separates the two --
%  results carry two components, sectional units three -- and is the
%  usual arrangement in a thesis of this length.
%  Added: notation, assumption, convention.
%======================================================================

\newtheorem{result}{Result}[chapter]   % common counter named 'result'

\theoremstyle{plain}                   % italic body — theorems, lemmas, etc.
\newtheorem{theorem}[result]{Theorem}
\newtheorem{lemma}[result]{Lemma}
\newtheorem{proposition}[result]{Proposition}
\newtheorem{corollary}[result]{Corollary}

\theoremstyle{definition}              % upright body — definitions and examples
\newtheorem{definition}[result]{Definition}
\newtheorem{example}[result]{Example}
\newtheorem{convention}[result]{Convention}
\newtheorem{assumption}[result]{Assumption}

\theoremstyle{remark}                  % upright body + italic header — remarks
\newtheorem{remark}[result]{Remark}
\newtheorem{notationbox}[result]{Notation}

% Unnumbered variant, for restating a result proved elsewhere
\theoremstyle{plain}
\newtheorem*{theorem*}{Theorem}

% Unnumbered convention, for the two global conventions stated before Section 1.1.
% These are standing conventions rather than results, and are never cross-referenced
% by number, so they are deliberately left out of the 'result' sequence.
\theoremstyle{definition}
\newtheorem*{convention*}{Convention}

%======================================================================
%  End-of-statement / end-of-proof markers.  See Convention 1.2:
%  hollow = quoted without proof, filled = end of an argument.
%
%  There are two situations and they need different macros.
%
%  (1) After a DISPLAY.  Writing a bare "\hfill $\square$" after
%      \end{equation} opens a fresh paragraph, so the marker lands on a
%      line of its own with a full \parskip above it and reads as
%      stranded.  \qedopen / \qedfilled pull it back up so that it sits
%      snugly under the closing display, still flush right.
%
%  (2) After running TEXT.  A bare "\hfill $\square$" is wrong here too,
%      but for a subtler reason: if the last line of the paragraph is
%      full, TeX breaks the line BEFORE the marker, the \hfill is
%      absorbed into the line that was just broken, and the symbol is
%      left alone at the LEFT margin of a new line.  (This happened to
%      Theorem 1.91.)  \qedopentext / \qedfilledtext use the standard
%      amsthm trick -- an empty \hbox placed before \nobreak\hfill, so
%      that any break happens ahead of the glue and the symbol carries
%      its \hfill with it -- and therefore sit flush right whether they
%      stay on the last line or move to a new one.
%
%  Usage:  ... \end{equation}
%          \qedopen                       % display, quoted result
%          \qedfilled                     % display, end of argument
%  and     ... and the claim follows. \qedfilledtext
%          ... see \cite{foo}. \qedopentext
%======================================================================
\makeatletter
% (1) after a display: pull the marker up under the last line of it
\newcommand{\thesis@endmark}[1]{%
  \par\nobreak
  \vskip -\parskip
  \vskip -0.9\baselineskip
  \noindent\hfill$#1$%
  \par
}
\newcommand{\qedopen}{\thesis@endmark{\square}}
\newcommand{\qedfilled}{\thesis@endmark{\blacksquare}}

% (2) after running text: never orphaned at the left margin
\newcommand{\thesis@textmark}[1]{%
  \leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill\quad$#1$%
}
\newcommand{\qedopentext}{\thesis@textmark{\square}}
\newcommand{\qedfilledtext}{\thesis@textmark{\blacksquare}}
\makeatother
