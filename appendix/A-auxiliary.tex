\chapter{Auxiliary Results}\label{app:auxiliary}

This appendix collects results which are used in the main text but whose statements
would have interrupted the argument there. Nothing here is original, and proofs are given
only where they run to a line or two; references are given in every case. The organising principle is simple: a
result belongs in the body if the argument cannot be followed without it, and belongs
here if the argument merely \emph{invokes} it. Results which we do prove, but whose proofs are too
long to sit comfortably in the body, are deferred instead to Appendix \ref{app:proofs}.

\section{Complements on conditional expectation}\label{app:condexp}

Throughout, $\mathcal{G} \subseteq \mathcal{F}$ is a sub-$\sigma$-algebra of a complete probability
space $\PSpace$, and all identities are understood to hold almost surely. Proofs may be found in
\cite{billingsley1995} or \cite{durrett2019}. Two further properties, the conditional Jensen
inequality and the $L^2$ projection characterisation, are used constantly in the main text and are
stated there as Proposition \ref{prop:condjensenl2}, to which we refer for them rather than
repeating them here.

\begin{proposition}[Properties of conditional expectation]\label{prop:condexpproperties}
Let $X, Y \in L^1(\Omega)$. Then:
\begin{itemize}
    \item[1.] $\mathbb{E}\!\left[\,\mathbb{E}[X \mid \mathcal{G}]\,\right] = \mathbb{E}[X]$; that is, $\mathbb{E}[X\mid\mathcal{G}]$ and $X$ have the same expectation.
    \item[2.] If $X$ is $\mathcal{G}$-measurable, then $\mathbb{E}[X \mid \mathcal{G}] = X$.
    \item[3.] If $X$ and $\mathcal{G}$ are independent --- that is, $\{X \in U\}$ and $A$ are independent events for every Borel $U \subseteq \mathbb{R}$ and every $A \in \mathcal{G}$ --- then $\mathbb{E}[X \mid \mathcal{G}] = \mathbb{E}[X]$.
    \item[4.] (\textbf{Taking out what is known}) If $Y$ is $\mathcal{G}$-measurable and $\mathbb{E}[|XY|]<\infty$, then $\mathbb{E}[XY \mid \mathcal{G}] = Y\,\mathbb{E}[X \mid \mathcal{G}]$.
    \item[5.] (\textbf{Tower property}) If $\mathcal{H}$ is a sub-$\sigma$-algebra of $\mathcal{G}$, then $\mathbb{E}[X \mid \mathcal{H}] = \mathbb{E}\!\left[\,\mathbb{E}[X \mid \mathcal{G}] \,\middle|\, \mathcal{H}\,\right]$.
    \item[6.] If $X \leq Y$, then $\mathbb{E}[X \mid \mathcal{G}] \leq \mathbb{E}[Y \mid \mathcal{G}]$; consequently $\left| \mathbb{E}[X \mid \mathcal{G}] \right| \leq \mathbb{E}[\,|X| \mid \mathcal{G}]$.
    \item[7.] For all $a,b \in \mathbb{R}$, $\mathbb{E}[aX + bY \mid \mathcal{G}] = a\,\mathbb{E}[X \mid \mathcal{G}] + b\,\mathbb{E}[Y \mid \mathcal{G}]$. In particular $\mathbb{E}[\,\cdot \mid \mathcal{G}]$ is a bounded linear operator from $L^{1}(\Omega,\mathcal{F})$ into $L^{1}(\Omega,\mathcal{G})$, a fact used to pass to limits in Theorem \ref{theorem:martingale_property}.
\end{itemize}
\hfill $\square$
\end{proposition}

The convergence theorems also carry over to the conditional setting, in exactly the form one would
expect.

\begin{proposition}[Conditional Fatou lemma]\label{prop:condfatou}
Let $X_n \geq 0$ with $X_n \in L^1(\Omega)$ for every $n \geq 1$, and suppose
$\liminf_{n\to\infty} X_n \in L^1(\Omega)$. Then
\[
\mathbb{E}\Bigl[\liminf_{n \to \infty} X_n \,\Big|\, \mathcal{G}\Bigr]
\leq \liminf_{n \to \infty} \mathbb{E}[X_n \mid \mathcal{G}] \qquad \text{almost surely.}
\]
\qedopen
\end{proposition}

\begin{proposition}[Conditional monotone convergence]\label{prop:condmct}
Let $0 \leq X_1 \leq X_2 \leq \cdots$ and suppose $X = \lim_{n\to\infty} X_n \in L^1(\Omega)$. Then
$\mathbb{E}[X \mid \mathcal{G}] = \lim_{n \to \infty} \mathbb{E}[X_n \mid \mathcal{G}]$ almost surely.
\hfill $\square$
\end{proposition}

\begin{proposition}[Conditional dominated convergence]\label{prop:conddct}
Suppose $X_n \to X$ almost surely and $|X_n| \leq Y$ for every $n$, with $Y \in L^1(\Omega)$.
Then $\mathbb{E}[X_n \mid \mathcal{G}] \to \mathbb{E}[X \mid \mathcal{G}]$ almost surely and in $L^1(\Omega)$.
\hfill $\square$
\end{proposition}

\section{Characteristic functions}\label{app:charfn}

Recall that the characteristic function of a real random variable $X$ is
$\charf_X(u) = \mathbb{E}[e^{iuX}]$, $u \in \mathbb{R}$. The following elementary
properties are used without comment in the main text.

\begin{proposition}\label{prop:charfnproperties}
For any real random variable $X$:
\begin{itemize}
    \item[i)] $\charf_X(0) = 1$ and $|\charf_X(u)| \leq 1$ for all $u \in \mathbb{R}$;
    \item[ii)] $\charf_X$ is uniformly continuous on $\mathbb{R}$;
    \item[iii)] $\charf_X(-u) = \overline{\charf_X(u)}$ (Hermitian symmetry);
    \item[iv)] $\charf_{aX+b}(u) = e^{iub}\charf_X(au)$ for all $a,b \in \mathbb{R}$;
    \item[v)] if $X$ and $Y$ are independent, then $\charf_{X+Y}(u) = \charf_X(u)\charf_Y(u)$;
    \item[vi)] if $\mathbb{E}[|X|^n] < \infty$, then $\charf_X$ is $n$ times differentiable and
    \[
    \charf_X^{(k)}(0) = i^k\,\mathbb{E}[X^k], \qquad 1 \leq k \leq n.
    \]
    Conversely, the existence of $2n$ continuous derivatives of $\charf_X$ in a neighbourhood of
    the origin implies $\mathbb{E}[|X|^{2n}] < \infty$.
\end{itemize}
\hfill $\square$
\end{proposition}

Property ii) deserves a word, since it is the reason the continuity hypothesis in the converse
half of Lévy's continuity theorem (Theorem \ref{theorem:LevyContinuity}) is a genuine restriction
rather than an automatic one: it constrains the \emph{limit} function, which is not assumed in
advance to be a characteristic function.

\section{Moment indeterminacy of the log-normal law}\label{app:heyde}

The following construction, due to Heyde \cite{heyde1963}, is referred to in Subsection
\ref{subsec:FundamentalDistr} and shows that a distribution possessing moments of all orders need not be
determined by them.

\begin{proposition}[Heyde]\label{prop:heyde}
Let $f_X$ denote the density of the standard log-normal law, that is, the density of $e^{Z}$ with
$Z \sim \mathcal{N}(0,1)$. For $a \in [-1,1]$ define
\begin{equation}\label{eq:heyde}
f_a(x) = f_X(x)\bigl(1 + a \sin(2\pi \log x)\bigr), \qquad x>0.
\end{equation}
Then every $f_a$ is a probability density on $(0,\infty)$, the laws are mutually distinct for
distinct $a$, and all of them share the moment sequence of the log-normal distribution.
\end{proposition}

\textbf{\textit{Proof:}} Non-negativity holds because $|a\sin(\cdot)| \leq 1$, and the moment identity reduces to showing that
\[
\int_0^\infty x^k f_X(x)\sin(2\pi \log x)\,dx = 0, \qquad k \in \mathbb{N}.
\]
Substituting $\log x = t + k$ turns the integrand into an odd function of $t$ multiplied by the
Gaussian weight, and the integral vanishes. Taking $k=0$ shows in particular that each $f_a$
integrates to $1$. \hfill $\blacksquare$

It is worth being precise about the logical status of this. Carleman's condition
$\sum_k m_{2k}^{-1/2k} = \infty$ is \emph{sufficient} for a law to be determined by its moments, and
the log-normal moments, growing like $e^{k^2/2}$, violate it. A violated sufficient condition
establishes only that the criterion is silent, never that the conclusion is false: what settles the
matter here is Proposition \ref{prop:heyde} itself, which exhibits the competing densities
explicitly. The growth rate merely explains why no elementary criterion could have helped. See
\cite{durrett2019} for the general theory.

\section{Grönwall's inequality}\label{app:gronwall}

The following is the standard comparison tool for integral inequalities. It is purely
deterministic, and in the stochastic setting it is applied pathwise or to expectations.

\begin{lemma}[Grönwall's inequality]\label{lem:gronwall}
Let $g, u \in L^1[0,T]$ and $\lambda > 0$ satisfy
\[
u(t) \leq g(t) + \lambda \int_0^t u(s)\,ds, \qquad 0 \leq t \leq T.
\]
Then
\[
u(t) \leq g(t) + \lambda \int_0^t g(s)\,e^{\lambda(t-s)}\,ds, \qquad 0 \leq t \leq T.
\]
In particular, if $g \equiv \kappa$ is constant, then $u(t) \leq \kappa e^{\lambda t}$ for all
$0 \leq t \leq T$; and if $\kappa = 0$, then $u \equiv 0$.
\hfill $\square$
\end{lemma}

The final sentence is the one actually used. Uniqueness arguments for differential equations
almost invariably reduce to showing that a suitable distance between two candidate solutions
satisfies a Grönwall inequality with $\kappa = 0$, whence it vanishes identically.

\section{Radon--Nikodym and the Jordan decomposition}\label{app:radonnikodym}

For completeness we record the decomposition underlying Corollary
\ref{cor:radon-nikodym-signed}.

\begin{theorem}[Jordan decomposition]\label{thm:jordan}
Every finite signed measure $\nu$ on a measurable space $(\Omega,\mathcal{F})$ admits a unique
decomposition $\nu = \nu^+ - \nu^-$ into mutually singular finite positive measures. The measure
$|\nu| := \nu^+ + \nu^-$ is called the total variation of $\nu$, and $\nu \ll \lambda$ if and only
if $|\nu| \ll \lambda$.
\hfill $\square$
\end{theorem}

See \cite{roydenfitzpatrick2010} for a proof.


\section{Uniform integrability and the class (D)}\label{app:ui}

Two integrability notions are invoked in the main text --- in the hypotheses of Theorem
\ref{thm:optionalstopping} and of Theorem \ref{thm:snell} respectively --- and are recorded here.

\begin{definition}[Uniform integrability]\label{def:ui}
A family $\{X_i\}_{i \in I}$ of integrable random variables is \textbf{uniformly integrable} if
\begin{equation}\label{eq:ui}
\lim_{c \to \infty}\ \sup_{i \in I}\ \E\bigl[\,|X_i|\,\Ind_{\{|X_i| > c\}}\,\bigr] \;=\; 0 .
\end{equation}
\end{definition}

The condition says that no member of the family carries a non-negligible amount of its mass
arbitrarily far out in the tail, \emph{uniformly in $i$}. It is strictly stronger than
$\sup_i \E|X_i| < \infty$ and strictly weaker than domination by a single integrable variable: if
$|X_i| \leq Y$ for all $i$ with $Y \in L^1$, then \eqref{eq:ui} follows from dominated convergence,
which is why hypothesis iii) of Theorem \ref{thm:optionalstopping} implies hypothesis ii). Its role
is that it is exactly the extra assumption converting convergence in probability into convergence in
$L^1$, and exactly what is needed for a martingale to be closed by a terminal variable. See
\cite[Ch.\ 4]{durrett2019}.

\begin{definition}[Class (D)]\label{def:classD}
A process $X$ is of \textbf{class (D)} if the family $\{X_\tau : \tau \text{ a stopping time}\}$,
indexed by all stopping times of the filtration, is uniformly integrable.
\end{definition}

The class is named for Doob. It is the hypothesis under which the Doob--Meyer decomposition
(Theorem \ref{thm:doobmeyer}) produces an \emph{integrable} increasing process rather than merely a
local one, and under which the Snell envelope of Theorem \ref{thm:snell} is well defined. Any
process dominated by an integrable variable is of class (D); in particular, so is any bounded
process, which covers every application in this dissertation.

\section{The essential supremum of a family of random variables}\label{app:esssup}

\begin{definition}[Essential supremum]\label{def:esssup}
Let $\{X_i\}_{i \in I}$ be a family of random variables on $\PSpace$, indexed by an arbitrary and
possibly uncountable set $I$. An \textbf{essential supremum} of the family, written
$\operatorname*{ess\,sup}_{i \in I} X_i$, is a random variable $X^*$ such that
\begin{itemize}
    \item[i)] $X_i \leq X^*$ almost surely, for every $i \in I$; and
    \item[ii)] if $Y$ satisfies i) in place of $X^*$, then $X^* \leq Y$ almost surely.
\end{itemize}
Such an $X^*$ exists and is unique up to null sets.
\end{definition}

The construction, for which see \cite[Appendix A]{karatzasshreve1991}, produces $X^*$ as the
pointwise supremum of a \emph{countable} subfamily, and that is the whole point of the notion. The
pointwise supremum $\sup_{i \in I} X_i(\omega)$ over an uncountable $I$ need not be measurable, and
even when it is it may exceed every $X_i$ on a set of positive probability in a way that carries no
information --- an uncountable union of null sets need not be null, which is the same phenomenon
that separates modification from indistinguishability in Remark \ref{rem:modification}. The
essential supremum repairs this by taking the supremum in the almost-sure order rather than
pointwise, and it is the correct object in \eqref{eq:optimalstopping}, where the family is indexed
by the uncountably many stopping times available to the holder of an American option.
