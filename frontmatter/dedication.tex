  \begin{dedication}
    Aus dem Paradies, das Cantor uns geschaffen hat, soll uns niemand vertreiben können. 
    \par   %% or a blank line
    \vspace{2\baselineskip}
    We shall never be expelled from the paradise Cantor created for us.

    \vspace{\baselineskip}
    
    \textbf{David Hilbert}
  \end{dedication}
